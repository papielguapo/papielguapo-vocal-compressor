cmake_minimum_required(VERSION 3.22)
project(PapisCompressor VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_PATH "C:/JUCE" CACHE PATH "JUCE source")
add_subdirectory("${JUCE_PATH}" "${CMAKE_BINARY_DIR}/JUCE")
juce_add_plugin(PapisCompressor COMPANY_NAME "Papielguapo" BUNDLE_ID "com.papielguapo.papiscompressor"
 IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
 EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE
 PLUGIN_MANUFACTURER_CODE Papi PLUGIN_CODE PpC1 FORMATS VST3 Standalone PRODUCT_NAME "Papis Compressor")
target_sources(PapisCompressor PRIVATE Source/PluginProcessor.h Source/PluginProcessor.cpp Source/PluginEditor.h Source/PluginEditor.cpp)
target_compile_definitions(PapisCompressor PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(PapisCompressor PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
option(PAPIS_TESTS "Build DSP tests" ON)
if(PAPIS_TESTS)
 enable_testing()
 juce_add_console_app(PapisTests PRODUCT_NAME "PapisTests")
 target_sources(PapisTests PRIVATE Tests.cpp Source/PluginProcessor.cpp Source/PluginEditor.cpp)
 target_compile_definitions(PapisTests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_link_libraries(PapisTests PRIVATE juce::juce_audio_utils juce::juce_dsp)
 add_test(NAME DSP COMMAND PapisTests)
endif()
